Developer SDKs

Integrate Battery Telemetry in Minutes

Our robust REST and GraphQL endpoints allow your engineering teams to sync vehicle health metrics instantly.

IonCo_fleet_sync.js Node.js SDK v4.2 
// Initialize IonCo Fleet Telemetry Client
const { IonCoFleet } = require('@IonCo/sdk');
const fleet = new IonCoFleet({
  apiKey: process.env.IonCo_API_KEY,
  region: 'in-south-1'
});

async function monitorFleetNode(vehicleId) {
  const telemetry = await fleet.getBatteryTelemetry(vehicleId);
  if (telemetry.voltage < 11.8) {
    await fleet.dispatchEmergencyVan({ vehicleId, priority: 'HIGH' });
  }
}
